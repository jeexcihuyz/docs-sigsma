API Documentation

Panduan lengkap unofficial API scrapping untuk layanan temporary email. Dokumen ini dirancang khusus untuk mem-bypass keamanan tingkat tinggi seperti Cloudflare Turnstile dan sistem token dinamis Javascript.

🛡️ Pedoman Browser Spoofing

Keamanan web masa kini menganalisis TLS Fingerprint dan HTTP/2 Headers. Request standar menggunakan library bawaan (seperti requests Python atau axios murni) akan 100% diblokir atau dimintai Captcha.

⚠️ Wajib Implementasi untuk Yopmail & Generator

Bagi Pengguna Python: Wajib menggunakan library curl_cffi (bukan requests atau httpx). Selalu sisipkan parameter impersonate="chrome120" di dalam AsyncSession() agar bot lu disangka Chrome asli.

Bagi Pengguna Node.js: Wajib menyusun Headers secara lengkap (termasuk Sec-CH-UA) dan gunakan library seperti cloudscraper atau got-tls untuk memanipulasi HTTP Ciphers.

1. Generator.email API

Generator.email menggunakan struktur URL sederhana namun dilindungi ketat oleh Cloudflare. Untuk membaca pesan, kita harus meniru perilaku browser dengan menyuntikkan cookie dan mengklik ID Pesan secara spesifik.

GET https://generator.email/{domain}/{username}
💡 Alur Logika By-Pass

Suntik Cookie surl 👉 Fetch URL Utama 👉 Cari Regex cur_msg_id 👉 Fetch URL Utama + ID Pesan 👉 Ekstrak DOM HTML.

import asyncio
import urllib.parse
import re
from bs4 import BeautifulSoup
from curl_cffi.requests import AsyncSession

class GeneratorEmailAPI:
    @staticmethod
    async def fetch_inbox(email: str):
        usr, dmn = email.split("@", 1)
        safe_u, safe_d = urllib.parse.quote(usr), urllib.parse.quote(dmn)
        base_url = f"https://generator.email/{safe_d}/{safe_u}"
        
        headers = {
            "Accept": "text/html,application/xhtml+xml,application/xml;q=0.9",
            "Cache-Control": "max-age=0",
            "Upgrade-Insecure-Requests": "1"
        }

        # 🛡️ SPOOFING: Impersonate Chrome 120
        async with AsyncSession(impersonate="chrome120", timeout=30.0) as client:
            client.cookies.set("surl", f"{safe_d}%2F{safe_u}", domain="generator.email")
            client.cookies.set("inbox_n", "1", domain="generator.email")
            
            # Request Index Inbox
            r = await client.get(base_url, headers=headers)
            if "just a moment" in r.text.lower() or r.status_code in [403, 503]:
                return {"error": "Terblokir Cloudflare WAF"}

            # Cari ID Pesan (Simulasi Klik)
            msg_id = None
            cur_msg = re.search(r'cur_msg_id:\s*"([^"]+)"', r.text)
            if cur_msg and len(cur_msg.group(1)) > 5:
                msg_id = cur_msg.group(1)
            else:
                paths = re.findall(r"loadInboxClientSide\('([^']+)'\)", r.text)
                if paths: msg_id = paths[0].split('/')[-1]

            if not msg_id: return {"error": "Inbox kosong"}

            # Request Buka Pesan
            r_msg = await client.get(f"{base_url}/{msg_id}", headers=headers)
            soup = BeautifulSoup(r_msg.text, "html.parser")
            
            # Ekstrak Payload Murni
            body = soup.find(class_="mess_bodiyy")
            return {"success": True, "raw_html": str(body) if body else r_msg.text}

const cloudscraper = require('cloudscraper');
const cheerio = require('cheerio');

async function fetchGeneratorInbox(email) {
    const [usr, dmn] = email.split('@');
    const baseUrl = `https://generator.email/${encodeURIComponent(dmn)}/${encodeURIComponent(usr)}`;
    
    // 🛡️ SPOOFING: Menggunakan cloudscraper default jar
    const scraper = cloudscraper.defaults({ jar: true });
    const headers = {
        'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9',
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) Chrome/120.0.0.0 Safari/537.36',
        'Cookie': `surl=${encodeURIComponent(dmn)}%2F${encodeURIComponent(usr)}; inbox_n=1;`
    };

    try {
        let res = await scraper.get({ uri: baseUrl, headers: headers, resolveWithFullResponse: true });
        let $ = cheerio.load(res.body);

        // Ekstrak MSG ID
        let msgId = null;
        const curMsgMatch = res.body.match(/cur_msg_id:\s*"([^"]+)"/);
        if (curMsgMatch && curMsgMatch[1].length > 5) {
            msgId = curMsgMatch[1];
        } else {
            const onClick = $('.list-group-item').first().attr('onclick');
            if (onClick) {
                const match = onClick.match(/'([^']+)'/);
                if (match) msgId = match[1].split('/').pop();
            }
        }

        if (!msgId) return { error: 'Inbox kosong' };

        // Fetch Pesan
        const msgRes = await scraper.get({ uri: `${baseUrl}/${msgId}`, headers: headers });
        $ = cheerio.load(msgRes);
        const htmlBody = $('.mess_bodiyy').html() || msgRes;

        return { success: true, raw_html: htmlBody };
    } catch (err) {
        return { error: err.message };
    }
}

2. Yopmail API

Yopmail tidak menggunakan Cloudflare Turnstile, melainkan menggunakan sistem Dymanic JS Token. Kita diwajibkan menarik token `yp` dari halaman depan, dan token `yj` dari file Javascript mereka untuk bisa memuat isi iFrame inbox.

GET https://yopmail.com/inbox
🚨 Peringatan Captcha Yopmail

Yopmail sangat agresif melacak IP Address dan TLS Request. Jika dalam satu proxy lu menembak API Yopmail berkali-kali tanpa delay, API akan merespon dengan halaman "CAPTCHA". Pastikan lu merotasi IP Proxy jika menggunakan script ini secara *Bulk*.

import datetime
import re
from bs4 import BeautifulSoup
from curl_cffi.requests import AsyncSession

class YopmailAPI:
    @staticmethod
    async def fetch_inbox(email: str):
        target = email.replace("@yopmail.com", "")
        
        # 🛡️ SPOOFING TLS
        async with AsyncSession(impersonate="chrome120", timeout=20.0) as client:
            client.headers.update({"Accept": "text/html,application/xhtml+xml,application/xml;q=0.9"})
            
            # 1. Bypass Index & Ambil Token YP
            r1 = await client.get("https://yopmail.com/")
            if "CAPTCHA" in r1.text.upper(): return {"error": "Terblokir Captcha Yopmail"}
            
            soup = BeautifulSoup(r1.text, "html.parser")
            token_yp = soup.find("input", {"id": "yp"}).get("value") if soup.find("input", {"id": "yp"}) else ""
            v_match = re.search(r"/ver/([\d\.]+)/", r1.text)
            versi_v = v_match.group(1) if v_match else "9.3"
            
            # 2. Ambil Token YJ dari Webmail.js
            await client.post("https://yopmail.com/", data={"yp": token_yp, "login": target})
            r_js = await client.get(f"https://yopmail.com/ver/{versi_v}/webmail.js")
            yj_match = re.search(r"yj=([a-zA-Z0-9]+)", r_js.text)
            token_yj = yj_match.group(1) if yj_match else ""
            
            # Set Cookie Waktu
            now = datetime.datetime.now()
            client.cookies.set("ytime", f"{now.hour}:{now.minute:02d}", domain="yopmail.com")
            
            # 3. Masuk ke iFrame Inbox
            params = {"login": target, "p": "1", "yp": token_yp, "yj": token_yj, "v": versi_v, "d": "", "ctrl": "", "id": "", "r_c": ""}
            r_inbox = await client.get("https://yopmail.com/inbox", params=params)
            soup_inbox = BeautifulSoup(r_inbox.text, "html.parser")
            
            email_div = soup_inbox.find("div", class_="m")
            if not email_div: return {"error": "Inbox kosong"}
            
            # 4. Buka Pesan Pertama
            id_baca = "m" + email_div.get("id")
            r_mail = await client.get("https://yopmail.com/mail", params={"b": target, "id": id_baca})
            soup_mail = BeautifulSoup(r_mail.text, "html.parser")
            
            body = soup_mail.find("div", id="mailmillieu")
            return {"success": True, "raw_html": str(body) if body else r_mail.text}

const cloudscraper = require('cloudscraper');
const cheerio = require('cheerio');

async function fetchYopmailInbox(email) {
    const target = email.replace('@yopmail.com', '');
    const scraper = cloudscraper.defaults({ jar: true });

    try {
        let res1 = await scraper.get('https://yopmail.com/');
        if (res1.includes('CAPTCHA')) throw new Error("Terkena Captcha Index");

        let $ = cheerio.load(res1);
        const token_yp = $('#yp').val();
        let versi_v = "9.3";
        const vMatch = res1.match(/\/ver\/([\d\.]+)\//);
        if (vMatch) versi_v = vMatch[1];

        await scraper.post({ uri: 'https://yopmail.com/', form: { yp: token_yp, login: target } });
        let resJs = await scraper.get(`https://yopmail.com/ver/${versi_v}/webmail.js`);
        
        const yjMatch = resJs.match(/yj=([a-zA-Z0-9]+)/);
        const token_yj = yjMatch ? yjMatch[1] : '';

        const d = new Date();
        const ytime = `${d.getHours()}:${String(d.getMinutes()).padStart(2, '0')}`;
        
        const qs = `login=${target}&p=1&yp=${token_yp}&yj=${token_yj}&v=${versi_v}&d=&ctrl=&id=&r_c=`;
        let resInbox = await scraper.get({
            uri: `https://yopmail.com/inbox?${qs}`,
            headers: { 'Referer': 'https://yopmail.com/wm', 'Cookie': `ytime=${ytime};` }
        });

        $ = cheerio.load(resInbox);
        const firstEmail = $('div.m').first();
        if (firstEmail.length === 0) return { error: 'Inbox Kosong' };

        const msgId = 'm' + firstEmail.attr('id');
        let resMail = await scraper.get({
            uri: `https://yopmail.com/mail?b=${target}&id=${msgId}`,
            headers: { 'Referer': 'https://yopmail.com/' }
        });

        $ = cheerio.load(resMail);
        const htmlBody = $('#mailmillieu').html() || resMail;

        return { success: true, raw_html: htmlBody };
    } catch (err) {
        return { error: err.message };
    }
}
